$* DECLARACIONES *$
DECVAR
numero : INT
cont : INT
a1 : FLOAT
b2 : INT
texto : STRING
hexVar : INT
valorReal : FLOAT
cadenaLarga : STRING
ENDDECVAR

PROGRAM.SECTION
$* ASIGNACIONES SIMPLES *$
numero = 2
cont = 0
a1 = 25.75
b2 = 100
texto = "Hola Mundo"
hexVar = 0hFF
valorReal = 99999.99
cadenaLarga = "@sdADaSjfla%dfg"

$* FOR *$
FOR al IS RANGE [numero * 2 + 8] {STEP 2}
    cont = cont + 1
NEXT al

$* DECISIONES *$
IF (numero > 5)
    texto = "El numero es mayor a 5"
ELSE
    texto = "El numero es menor o igual a 5"
ENDIF
